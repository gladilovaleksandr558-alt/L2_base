function bingo(ticket, win){
  let miniWins = 0;
  
  for (let i = 0; i < ticket.length; i++) {
    let word = ticket[i][0];
    let code = ticket[i][1];
    
    let hasMatch = false;
    for (let j = 0; j < word.length; j++) {
      if (word.charCodeAt(j) === code) {
        hasMatch = true;
      }
    }
    
    if (hasMatch) {
      miniWins++;
    }
  }
  
  if (miniWins >= win) {
    return 'Winner!';
  } else {
    return 'Loser!';
  }
}
