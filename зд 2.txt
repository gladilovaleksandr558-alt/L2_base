function ipv4Parser(ip, mask){
    const ipToLong = str => str.split('.').reduce((acc, octet) => (acc << 8) + parseInt(octet, 10), 0) >>> 0;
    
    const longToIp = long => [
        (long >>> 24) & 255,
        (long >>> 16) & 255,
        (long >>> 8) & 255,
        long & 255
    ].join('.');

    const ipLong = ipToLong(ip);
    const maskLong = ipToLong(mask);

    return [
        longToIp(ipLong & maskLong),
        longToIp((ipLong & ~maskLong) >>> 0)
    ];
}
